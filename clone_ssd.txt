#!/bin/bash
set -euo pipefail

# ============================================================
#  Clone HDD -> SSD  (Intel NUC6CAYH)
#  Layout SSD (238.5G):
#    sdb1  976M   vfat   /boot/efi
#    sdb2  ~31.5G ext4   /
#    sdb3  ~11.8G ext4   /var
#    sdb4  ~3.9G  swap
#    sdb5  ~2.8G  ext4   /tmp
#    sdb6  sisa   ext4   /home
# ============================================================

HDD=/dev/sda
SSD=/dev/sdb

# ---- Install dependensi yang kurang ----
if [ "$(id -u)" != 0 ]; then
    echo "Jalankan dengan sudo: sudo bash clone_ssd.sh"
    exit 1
fi
echo ">>> Memastikan tool yang dibutuhkan terinstal ..."
apt-get update -y
DEBIAN_FRONTEND=noninteractive apt-get install -y \
    parted gdisk dosfstools rsync grub-efi-amd64 efibootmgr

# ---- WARNING ----
echo "======================================================"
echo " PERINGATAN: Semua data di $SSD AKAN DIHAPUS TOTAL!"
echo " HDD asal tetap aman di $HDD (tidak disentuh kecuali dibaca)."
echo "======================================================"
read -p "Ketik 'YES' untuk lanjut: " confirm
if [ "$confirm" != "YES" ]; then
    echo "Dibatalkan."
    exit 1
fi

# Pastikan device target benar (2.5\" SATA SSD 256GB)
echo ""
echo "Konfirmasi disk target:"
lsblk -o NAME,SIZE,MODEL $SSD
read -p "Ini benar SSD 256GB (bukan $HDD HDD 500GB)? [y/N]: " ok
if [ "$ok" != "y" ] && [ "$ok" != "Y" ]; then
    echo "Dibatalkan."
    exit 1
fi

set -x

# ---- 0. Bersihkan SSD dari mount/swap lama yang mungkin tersisa ----
echo ">>> Membersihkan mount & swap lama di $SSD ..."
# Matikan swap pada partisi SSD bila aktif
for p in ${SSD}*; do
    if swapon --show=NAME --noheadings 2>/dev/null | grep -q "^$p$"; then
        swapoff "$p" 2>/dev/null || true
    fi
done
# Unmount semua partisi SSD (lewat mountpoint staging bila ada)
# Catatan: hanya unmount lewat mountpoint. Jangan panggil findmnt --source
# pada device ini; pada host khusus, itu bisa mem-block/macet.
STAGE=/mnt/ssdclone
umount -R "$STAGE" 2>/dev/null || true
sleep 1

# ---- 1. Hapus seluruh partisi di SSD, buat tabel GPT ----
wipefs -a "$SSD"
parted -s "$SSD" mklabel gpt

# ---- 2. Buat partisi baru (mengikuti ukuran partisi HDD semula) ----
# sdb1: EFI 976M (align 1MiB)
parted -s "$SSD" mkpart "EFI" fat32 1MiB 977MiB
parted -s "$SSD" set 1 esp on

# sdb2: root 31.5G
parted -s "$SSD" mkpart "root" ext4 977MiB 32255MiB

# sdb3: var 11.8G
parted -s "$SSD" mkpart "var" ext4 32255MiB 44338MiB

# sdb4: swap 3.9G
parted -s "$SSD" mkpart "swap" linux-swap 44338MiB 48331MiB

# sdb5: tmp 2.8G
parted -s "$SSD" mkpart "tmp" ext4 48331MiB 51198MiB

# sdb6: home = sisa (sampai akhir disk)
parted -s "$SSD" mkpart "home" ext4 51198MiB 100%

# ---- 3. Refresh kernel untuk melihat partisi baru ----
partprobe "$SSD"
sleep 2

# ---- 4. Buat filesystem ----
mkfs.vfat -F32 -n "EFI" "${SSD}1"
mkfs.ext4 -F -L "root" "${SSD}2"
mkfs.ext4 -F -L "var" "${SSD}3"
mkswap "${SSD}4"
mkfs.ext4 -F -L "tmp" "${SSD}5"
mkfs.ext4 -F -L "home" "${SSD}6"

# ---- 5. Mount layout staging ----
STAGE=/mnt/ssdclone
mkdir -p $STAGE
mount "${SSD}2" $STAGE
mkdir -p $STAGE/var $STAGE/tmp $STAGE/home $STAGE/boot
mount "${SSD}3" $STAGE/var
mount "${SSD}5" $STAGE/tmp
mount "${SSD}6" $STAGE/home
mkdir -p $STAGE/boot/efi
mount "${SSD}1" $STAGE/boot/efi

echo ""
echo "=== Mount layout staging selesai ==="
lsblk -o NAME,SIZE,MOUNTPOINT $SSD
echo ""
echo "Folder staging: $STAGE"
echo "Layout partisi SSD sudah siap. LANJUTKAN ke langkah copy data."
read -p "Tekan Enter untuk lanjut ke copy data (rsync)..."

# ---- 6. Copy data dari HDD (dari direktori yang sudah ter-mount) ----
# CATATAN: /var, /tmp, /home, /boot/efi sudah ter-mount live sebagai partisi
# terpisah di HDD. Karena itu jangan copy lagi di dalam rsync "/" (dedupe -x),
# dan copy masing-masing langsung dari path hidup ke staging.

echo ">>> Copy / (root, tanpa /var /tmp /home /boot) ..."
rsync -aHAXx --exclude=/var --exclude=/tmp --exclude=/home --exclude=/boot/efi / $STAGE/

echo ">>> Copy /var -> ssd/var ..."
rsync -aHAXx /var/ $STAGE/var/

echo ">>> Copy /tmp -> ssd/tmp ..."
rsync -aHAXx /tmp/ $STAGE/tmp/

echo ">>> Copy /home -> ssd/home ..."
rsync -aHAXx /home/ $STAGE/home/

echo ">>> Copy /boot/efi -> ssd/boot/efi ..."
rsync -aHAXx /boot/efi/ $STAGE/boot/efi/

# ---- 7. Perbaiki fstab di SSD dengan UUID baru ----
echo ">>> Update /etc/fstab di SSD ..."
NEWFSTAB=$STAGE/etc/fstab
cat > /tmp/fstab.new <<EOF
# Generated by clone_ssd.sh
UUID=$(blkid -s UUID -o value ${SSD}2) /               ext4    errors=remount-ro 0       1
UUID=$(blkid -s UUID -o value ${SSD}1) /boot/efi       vfat    umask=0077      0       1
UUID=$(blkid -s UUID -o value ${SSD}6) /home           ext4    defaults        0       2
UUID=$(blkid -s UUID -o value ${SSD}5) /tmp            ext4    defaults        0       2
UUID=$(blkid -s UUID -o value ${SSD}3) /var            ext4    defaults        0       2
UUID=$(blkid -s UUID -o value ${SSD}4) none            swap    sw              0       0
EOF
cp "$NEWFSTAB" /tmp/fstab.old.bak
cp /tmp/fstab.new "$NEWFSTAB"

echo ""
echo "=== Copy data selesai ==="
echo "fstab lama tersimpan di /tmp/fstab.old.bak"
read -p "Tekan Enter untuk lanjut ke instalasi GRUB (bootloader) ke SSD..."

# ---- 8. Chroot ke SSD dan instal GRUB ----
mount --bind /dev $STAGE/dev
mount --bind /dev/pts $STAGE/dev/pts
mount --bind /proc $STAGE/proc
mount --bind /sys $STAGE/sys
mount --bind /sys/firmware/efi/efivars $STAGE/sys/firmware/efi/efivars

# Bersihkan GRUB lama di SSD lalu instal ulang ke SSD
chroot $STAGE grub-install --target=x86_64-efi --efi-directory=/boot/efi --bootloader-id=GRUB --recheck
chroot $STAGE update-grub

# ---- 9. Update efibootmgr agar boot dari SSD ----
efibootmgr --create --disk "$SSD" --part 1 --label "SSD-GRUB" --loader '\EFI\GRUB\grubx64.efi' || echo "efibootmgr create gagal/lewati (NUC default boot)"

echo ""
echo "======================================================"
echo " CLONE SELESAI!"
echo "======================================================"
echo ""
echo " Yang perlu dilakukan SELANJUTNYA:"
echo "  1. sudo umount -R $STAGE"
echo "  2. Matikan laptop: sudo poweroff"
echo "  3. Lepas SSD dari enclosure USB"
echo "  4. Buka NUC, cabut HDD, pasang SSD di slot 2.5\" SATA"
echo "  5. Nyalakan. GRUB akan boot dari SSD."
echo "  6. Jika tidak boot, masuk BIOS (F2) & pilih SSD sebagai boot."
echo "  7. Setelah boot OK, verifikasi: lsblk & df -h"
echo ""
echo " NOTE: HDD lama TIDAK diubah data-nya (baca-saja)."
echo "       Jika semua bekerja, HDD bisa diformat ulang nanti."
